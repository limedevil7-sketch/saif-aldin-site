(() => { // تحميل الخطوط const fontLink = document.createElement('link'); fontLink.href = 'https://fonts.googleapis.com/css2?family=Cairo:wght@400;700&display=swap'; fontLink.rel = 'stylesheet'; document.head.appendChild(fontLink); // إعداد tailwind tailwind.config = { theme: { extend: { fontFamily: { Cairo: ['Cairo', 'sans-serif'], }, colors: { primary: '#3B82F6', }, }, }, }; // عناصر DOM const prayerTimeText = document.querySelector('.prayer-time'); const dateText = document.querySelector('.date'); const adkarText = document.querySelector('.adkar'); const refreshBtn = document.querySelector('.refresh'); const shareBtn = document.querySelector('.share'); let prayerTimes = null; let userLocation = null; // وظيفة لعرض التاريخ function displayCurrentDate() { const date = new Date(); dateText.innerHTML = formatArabicDate(date); } // تحميل أوقات الصلاة async function loadPrayerTimes() { try { const storedLocation = localStorage.getItem('location'); if (storedLocation) { userLocation = JSON.parse(storedLocation); } else { await getUserLocation(); } if (!userLocation) { alert('تعذر تحديد الموقع. يرجى تفعيل الموقع الجغرافي.'); return; } const { latitude, longitude } = userLocation; const response = await fetch(`https://api.aladhan.com/v1/timings/${new Date().getTime()}/?latitude=${latitude}&longitude=${longitude}&method=2`); const data = await response.json(); if (!data || data.code !== 200 || !data.data) { throw new Error('فشل في تحميل أوقات الصلاة'); } prayerTimes = data.data.timings; updatePrayerTimeDisplay(prayerTimes); } catch (error) { console.error('خطأ في تحميل أوقات الصلاة:', error); alert('حدث خطأ أثناء تحميل أوقات الصلاة.'); } } // الحصول على الموقع الجغرافي async function getUserLocation() { try { if (navigator.geolocation) { const position = await new Promise((resolve, reject) => { navigator.geolocation.getCurrentPosition(resolve, reject); }); userLocation = { latitude: position.coords.latitude, longitude: position.coords.longitude, }; localStorage.setItem('location', JSON.stringify(userLocation)); } else { alert('متصفحك لا يدعم تحديد الموقع الجغرافي.'); } } catch (error) { console.error('خطأ في تحديد الموقع:', error); } } // حساب الوقت المتبقي للصلاة function getTimeRemaining(prayerTimeStr) { const [hours, minutes] = prayerTimeStr.split(':').map(Number); const now = new Date(); const prayerDate = new Date(); prayerDate.setHours(hours, minutes, 0, 0); if (prayerDate.getTime() <= now.getTime()) { // إذا كانت الصلاة بعد الساعة الحالية غدا prayerDate.setDate(prayerDate.getDate() + 1); } const diffMs = prayerDate - now; const diffHrs = Math.floor(diffMs / (1000 * 60 * 60)); const diffMins = Math.floor((diffMs / (1000 * 60)) % 60); return { hours: diffHrs, minutes: diffMins }; } // عرض الوقت المتبقي حتى الصلاة function updatePrayerTimeDisplay(times) { if (prayerTimeText && times) { const currentTime = new Date().getTime(); const prayerNames = ['Fajr', 'Dhuhr', 'Asr', 'Maghrib', 'Isha']; let nextPrayer = null; for (let prayer of prayerNames) { const prayerTimeStr = times[prayer]; const { hours, minutes } = getTimeRemaining(prayerTimeStr); // إذا كانت الصلاة قادمة اليوم if (hours >= 0 && minutes >= 0) { nextPrayer = { name: prayer, time: prayerTimeStr, hours, minutes }; break; } } if (!nextPrayer) { // إذا كانت أوقات الصلاة قد انتهت، نعرض الفجر ليوم الغد const { hours, minutes } = getTimeRemaining(times['Fajr']); nextPrayer = { name: 'Fajr', time: times['Fajr'], hours, minutes }; } prayerTimeText.innerHTML = `باقي: ${nextPrayer.hours} ساعة و${nextPrayer.minutes} دقيقة حتى صلاة ${nextPrayer.name}`; } } // تنسيق التاريخ العربي function formatArabicDate(date) { const options = { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric' }; return date.toLocaleDateString('ar-EG', options); } // تحديث البيانات كل دقيقة لعرض الوقت المتبقي setInterval(() => { if (prayerTimes) { updatePrayerTimeDisplay(prayerTimes); } }, 60000); displayCurrentDate(); loadPrayerTimes(); // زر التحديث if (refreshBtn) { refreshBtn.addEventListener('click', loadPrayerTimes); } // زر المشاركة if (shareBtn) { shareBtn.addEventListener('click', () => { if (navigator.share) { navigator.share({ title: 'مواقيت الصلاة والأذكار', text: 'تفضل بزيارة تطبيق مواقيت الصلاة والأذكار.', url: window.location.href, }).catch(console.error); } else { alert('خاصية المشاركة غير مدعومة في هذا المتصفح.'); } }); } // عرض الأذكار function showAdkar(text) { if (adkarText) { adkarText.innerHTML = text; } } // استدعاءات مبدئية loadPrayerTimes(); displayCurrentDate();
